cmake_minimum_required(VERSION 3.22)
project(dronestock_btf_core VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(Eigen3 3.3 REQUIRED NO_MODULE)
find_path(BTF_JSON_INCLUDE_DIR nlohmann/json.hpp REQUIRED)
add_library(btf_core src/h80.cpp src/btf.cpp src/freeze.cpp)
set_target_properties(btf_core PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_include_directories(btf_core PUBLIC include)
target_link_libraries(btf_core PRIVATE Eigen3::Eigen)
if(MSVC)
  target_compile_options(btf_core PRIVATE /W4)
else()
  target_compile_options(btf_core PRIVATE -Wall -Wextra -Wpedantic)
endif()
add_executable(btf_replay src/replay.cpp)
target_include_directories(btf_replay PRIVATE ${BTF_JSON_INCLUDE_DIR})
target_link_libraries(btf_replay PRIVATE btf_core)
add_library(btf_native SHARED src/c_api.cpp)
target_include_directories(btf_native PRIVATE ${BTF_JSON_INCLUDE_DIR})
target_link_libraries(btf_native PRIVATE btf_core)
include(CTest)
if(BUILD_TESTING)
  add_executable(btf_tests tests/test_core.cpp)
  target_link_libraries(btf_tests PRIVATE btf_core)
  add_test(NAME btf_core_contract COMMAND btf_tests)
  find_package(Python3 COMPONENTS Interpreter QUIET)
  if(Python3_Interpreter_FOUND)
    add_test(NAME btf_replay_contract COMMAND ${Python3_EXECUTABLE}
      ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_replay.py
      --binary $<TARGET_FILE:btf_replay>
      --examples ${CMAKE_CURRENT_SOURCE_DIR}/examples)
    if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/../test/processing/test_measured_btf.py")
      add_test(NAME btf_native_adapter COMMAND ${Python3_EXECUTABLE}
        ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_adapter.py
        --library $<TARGET_FILE:btf_native> --package-root ${CMAKE_CURRENT_SOURCE_DIR}/..)
    endif()
  endif()
endif()
install(TARGETS btf_core btf_replay btf_native ARCHIVE DESTINATION lib LIBRARY DESTINATION lib RUNTIME DESTINATION bin)
install(DIRECTORY include/ DESTINATION include)
